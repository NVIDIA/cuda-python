20def _parse_version_triple(version_str: str) -> tuple[int, int, int]: 

21 """Parse a PEP 440 version string into a (major, minor, patch) triple. 

22  

23 Strips local-version identifiers and handles pre-release suffixes such as 

24 ``0b1`` or ``0rc1`` by extracting only the leading integer from each 

25 release segment. 

26 """ 

27 parts = version_str.partition("+")[0].split(".")[:3] 

28 ints = ([int(m.group(1)) if (m := re.match(r"(\d+)", v)) else 0 for v in parts] + [0, 0, 0])[:3] 

29 return (ints[0], ints[1], ints[2])

33def binding_version() -> tuple[int, int, int]: 

34 """Return the cuda-bindings version as a (major, minor, patch) triple.""" 

35 try: 

36 version_str = importlib.metadata.version("cuda-bindings") 

37 except importlib.metadata.PackageNotFoundError:

43def driver_version() -> tuple[int, int, int]:
