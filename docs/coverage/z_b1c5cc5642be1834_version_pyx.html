12def _parse_version_triple(version_str: str) -> tuple[int, int, int]: 

13 """Parse a PEP 440 version string into a (major, minor, patch) triple. 

14  

15 Strips local-version identifiers and handles pre-release suffixes such as 

16 ``0b1`` or ``0rc1`` by extracting only the leading integer from each 

17 release segment. 

18 """ 

19 parts = version_str.partition("+")[0].split(".")[:3] 

20 ints = ([int(m.group(1)) if (m := re.match(r"(\d+)", v)) else 0 for v in parts] + [0, 0, 0])[:3] 

21 return (ints[0], ints[1], ints[2])

25def binding_version() -> tuple[int, int, int]: 

26 """Return the cuda-bindings version as a (major, minor, patch) triple.""" 

27 try: 

28 version_str = importlib.metadata.version("cuda-bindings") 

29 except importlib.metadata.PackageNotFoundError: 

30 version_str = importlib.metadata.version("cuda-python") 

31 return _parse_version_triple(version_str)

35def driver_version() -> tuple[int, int, int]: 

36 """Return the CUDA driver version as a (major, minor, patch) triple.""" 

37 cdef int ver = handle_return(driver.cuDriverGetVersion()) 

38 return (ver // 1000, (ver // 10) % 100, ver % 10)
